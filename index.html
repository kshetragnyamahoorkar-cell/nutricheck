AI-Powered Stock Market Data Chatbot ==================================== Fetches statistical stock market data via Yahoo Finance (yfinance) and exports clean Excel workbooks. Usage: python stock_market_chatbot.py Examples of natural-language commands: - "history AAPL 1y" - "get historical data for MSFT and GOOGL for 6 months" - "fundamentals TSLA" - "income statement AAPL annual" - "balance sheet NVDA quarterly" - "dividends AAPL" - "compare AAPL MSFT GOOGL 1y" - "info META" - "help" - "quit" / "exit" Requirements (install once): pip install yfinance pandas openpyxl Note: Yahoo Finance has rate limits. If you hit them, wait a few minutes and try again. For heavy use, consider Alpha Vantage / Finnhub with an API key. """ from __future__ import annotations import re import sys from datetime import datetime from pathlib import Path from typing import List, Optional, Tuple import pandas as pd try: import yfinance as yf except ImportError: print("yfinance is required. Install with: pip install yfinance") sys.exit(1) # --------------------------------------------------------------------------- # Helpers # --------------------------------------------------------------------------- PERIOD_MAP = { "1d": "1d", "5d": "5d", "1w": "5d", "1m": "1mo", "1mo": "1mo", "3m": "3mo", "3mo": "3mo", "6m": "6mo", "6mo": "6mo", "1y": "1y", "2y": "2y", "5y": "5y", "10y": "10y", "ytd": "ytd", "max": "max", "1 year": "1y", "one year": "1y", "6 months": "6mo", "3 months": "3mo", "1 month": "1mo", "5 years": "5y", } def normalize_period(text: str) -> str: text = text.lower().strip() for k, v in PERIOD_MAP.items(): if k in text: return v return "1y" # sensible default def extract_tickers(text: str) -> List[str]: """Pull uppercase ticker-like tokens (1-5 letters, optionally .XX).""" # Common words to ignore stop = { "HISTORY", "HISTORICAL", "DATA", "FOR", "AND", "THE", "GET", "SHOW", "FETCH", "EXPORT", "SAVE", "TO", "EXCEL", "FUNDAMENTALS", "INFO", "INFORMATION", "INCOME", "STATEMENT", "BALANCE", "SHEET", "CASH", "FLOW", "DIVIDENDS", "SPLITS", "COMPARE", "COMPARISON", "ANNUAL", "QUARTERLY", "YEAR", "MONTH", "MONTHS", "WEEK", "WEEKS", "DAY", "DAYS", "HELP", "QUIT", "EXIT", "PLEASE", "ME", "ALL", "STATS", "STATISTICAL", "STOCK", "MARKET", "OF", "FROM", "WITH", } candidates = re.findall(r"\b([A-Z]{1,5}(?:\.[A-Z]{1,3})?)\b", text.upper()) tickers = [c for c in candidates if c not in stop] # Deduplicate while preserving order seen = set() unique = [] for t in tickers: if t not in seen: seen.add(t) unique.append(t) return unique def safe_filename(name: str) -> str: return re.sub(r"[^\w\-_.]", "_", name) # --------------------------------------------------------------------------- # Data fetchers # --------------------------------------------------------------------------- def fetch_history(tickers: List[str], period: str = "1y") -> dict[str, pd.DataFrame]: results = {} for ticker in tickers: try: t = yf.Ticker(ticker) df = t.history(period=period, auto_adjust=True) if df.empty: print(f" ⚠️ No price history for {ticker}") continue df = df.reset_index() # Clean column names df.columns = [c.replace(" ", "_") for c in df.columns] results[ticker] = df print(f" ✓ History {ticker}: {len(df)} rows ({period})") except Exception as e: print(f" ✗ Failed history {ticker}: {e}") return results def fetch_info(tickers: List[str]) -> dict[str, pd.DataFrame]: results = {} for ticker in tickers: try: t = yf.Ticker(ticker) info = t.info if not info: print(f" ⚠️ No info for {ticker}") continue # Flatten into a tidy key-value DataFrame rows = [{"Metric": k, "Value": str(v)} for k, v in sorted(info.items()) if v is not None] df = pd.DataFrame(rows) results[ticker] = df print(f" ✓ Info {ticker}: {len(df)} metrics") except Exception as e: print(f" ✗ Failed info {ticker}: {e}") return results def fetch_financials(tickers: List[str], statement: str = "income", freq: str = "yearly") -> dict[str, pd.DataFrame]: """ statement: income | balance | cashflow freq: yearly | quarterly """ results = {} for ticker in tickers: try: t = yf.Ticker(ticker) if statement == "income": df = t.income_stmt if freq == "yearly" else t.quarterly_income_stmt elif statement == "balance": df = t.balance_sheet if freq == "yearly" else t.quarterly_balance_sheet elif statement == "cashflow": df = t.cashflow if freq == "yearly" else t.quarterly_cashflow else: continue if df is None or df.empty: print(f" ⚠️ No {statement} data for {ticker}") continue # Transpose so dates become columns or keep as-is (yfinance already has dates as columns) results[ticker] = df print(f" ✓ {statement.title()} ({freq}) {ticker}: {df.shape}") except Exception as e: print(f" ✗ Failed {statement} {ticker}: {e}") return results def fetch_dividends(tickers: List[str]) -> dict[str, pd.DataFrame]: results = {} for ticker in tickers: try: t = yf.Ticker(ticker) s = t.dividends if s is None or s.empty: print(f" ⚠️ No dividends for {ticker}") continue df = s.reset_index() df.columns = ["Date", "Dividend"] results[ticker] = df print(f" ✓ Dividends {ticker}: {len(df)} payments") except Exception as e: print(f" ✗ Failed dividends {ticker}: {e}") return results def fetch_splits(tickers: List[str]) -> dict[str, pd.DataFrame]: results = {} for ticker in tickers: try: t = yf.Ticker(ticker) s = t.splits if s is None or s.empty: print(f" ⚠️ No splits for {ticker}") continue df = s.reset_index() df.columns = ["Date", "Split_Ratio"] results[ticker] = df print(f" ✓ Splits {ticker}: {len(df)} events") except Exception as e: print(f" ✗ Failed splits {ticker}: {e}") return results # --------------------------------------------------------------------------- # Excel exporter # --------------------------------------------------------------------------- def export_to_excel( data_dict: dict[str, pd.DataFrame], base_name: str, prefix: str = "stock_data", ) -> Path: """Write multiple DataFrames to a multi-sheet Excel file.""" timestamp = datetime.now().strftime("%Y%m%d_%H%M%S") filename = f"{prefix}_{safe_filename(base_name)}_{timestamp}.xlsx" path = Path(filename) with pd.ExcelWriter(path, engine="openpyxl") as writer: for sheet_name, df in data_dict.items(): # Excel sheet names max 31 chars safe_sheet = safe_filename(sheet_name)[:31] df.to_excel(writer, sheet_name=safe_sheet, index=True) print(f"\n✅ Saved → {path.resolve()}") return path # --------------------------------------------------------------------------- # Intent parser (simple rule-based "AI") # --------------------------------------------------------------------------- def parse_and_run(user_input: str) -> None: text = user_input.strip() if not text: return lower = text.lower() # Exit if lower in {"quit", "exit", "q", "bye"}: print("Goodbye! 📈") sys.exit(0) # Help if lower in {"help", "?", "h"}: print(__doc__) return tickers = extract_tickers(text) if not tickers: print("I couldn't find any ticker symbols. Try e.g. 'history AAPL 1y'") return period = normalize_period(text) # ---- History / OHLCV ---- if any(w in lower for w in ["history", "historical", "price", "ohlc", "ohlcv", "chart data"]): print(f"\nFetching price history for {', '.join(tickers)} (period={period})...") data = fetch_history(tickers, period) if data: export_to_excel(data, "_".join(tickers), prefix="history") # ---- Company info / key stats ---- elif any(w in lower for w in ["info", "information", "fundamentals", "stats", "profile", "overview"]): print(f"\nFetching company info / key stats for {', '.join(tickers)}...") data = fetch_info(tickers) if data: export_to_excel(data, "_".join(tickers), prefix="info") # ---- Financial statements ---- elif "income" in lower or "profit" in lower or "earnings" in lower: freq = "quarterly" if "quarter" in lower else "yearly" print(f"\nFetching income statement ({freq}) for {', '.join(tickers)}...") data = fetch_financials(tickers, "income", freq) if data: export_to_excel(data, "_".join(tickers), prefix=f"income_{freq}") elif "balance" in lower: freq = "quarterly" if "quarter" in lower else "yearly" print(f"\nFetching balance sheet ({freq}) for {', '.join(tickers)}...") data = fetch_financials(tickers, "balance", freq) if data: export_to_excel(data, "_".join(tickers), prefix=f"balance_{freq}") elif "cash" in lower and "flow" in lower: freq = "quarterly" if "quarter" in lower else "yearly" print(f"\nFetching cash flow ({freq}) for {', '.join(tickers)}...") data = fetch_financials(tickers, "cashflow", freq) if data: export_to_excel(data, "_".join(tickers), prefix=f"cashflow_{freq}") # ---- Dividends ---- elif "dividend" in lower: print(f"\nFetching dividends for {', '.join(tickers)}...") data = fetch_dividends(tickers) if data: export_to_excel(data, "_".join(tickers), prefix="dividends") # ---- Splits ---- elif "split" in lower: print(f"\nFetching stock splits for {', '.join(tickers)}...") data = fetch_splits(tickers) if data: export_to_excel(data, "_".join(tickers), prefix="splits") # ---- Compare (default to history) ---- elif "compare" in lower or "comparison" in lower: print(f"\nFetching comparison history for {', '.join(tickers)} (period={period})...") data = fetch_history(tickers, period) if data: # Also create a side-by-side Close prices sheet closes = {} for tkr, df in data.items(): if "Close" in df.columns and "Date" in df.columns: closes[tkr] = df.set_index("Date")["Close"] if closes: compare_df = pd.DataFrame(closes) data["Close_Comparison"] = compare_df export_to_excel(data, "_".join(tickers), prefix="compare") # ---- Default: treat as history request ---- else: print(f"\nAssuming price history request for {', '.join(tickers)} (period={period})...") print("(Tip: be more specific – try 'history', 'info', 'income statement', etc.)") data = fetch_history(tickers, period) if data: export_to_excel(data, "_".join(tickers), prefix="history") # --------------------------------------------------------------------------- # Main loop # --------------------------------------------------------------------------- def main() -> None: print("=" * 60) print(" 📈 AI-Powered Stock Market Data Chatbot") print(" Fetches live statistical data → Excel") print("=" * 60) print("Type a request (or 'help' / 'quit')\n") while True: try: user = input("You › ").strip() except (EOFError, KeyboardInterrupt): print("\nGoodbye!") break if not user: continue try: parse_and_run(user) except Exception as e: print(f"Unexpected error: {e}") print() if __name__ == "__main__": yfinance>=0.2.40 pandas>=2.0 openpyxl>=3.1 main()
